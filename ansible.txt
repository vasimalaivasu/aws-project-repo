---
# =====================================================
# COMMON SETUP (ALL EC2)
# =====================================================
- name: Common setup
  hosts: all
  become: yes
  tasks:
    - name: Install base packages
      apt:
        name:
          - ca-certificates
          - curl
          - gnupg
          - lsb-release
        state: present
        update_cache: yes

# =====================================================
# JENKINS EC2 (Jenkins + Docker + Maven)
# =====================================================
- name: Setup Jenkins EC2
  hosts: jenkins
  become: yes
  tasks:

    - name: Install Java 17
      apt:
        name: openjdk-17-jdk
        state: present

    # Jenkins Repo
    - name: Add Jenkins GPG key
      apt_key:
        url: https://pkg.jenkins.io/debian-stable/jenkins.io-2026.key
        state: present

    - name: Add Jenkins repo
      apt_repository:
        repo: deb https://pkg.jenkins.io/debian-stable binary/
        state: present

    # Docker Repo
    - name: Add Docker GPG key
      apt_key:
        url: https://download.docker.com/linux/ubuntu/gpg
        state: present

    - name: Add Docker repo
      apt_repository:
        repo: deb https://download.docker.com/linux/ubuntu {{ ansible_distribution_release }} stable
        state: present

    - name: Update apt
      apt:
        update_cache: yes

    - name: Install Jenkins, Docker, Maven
      apt:
        name:
          - jenkins
          - docker-ce
          - docker-ce-cli
          - containerd.io
          - maven
        state: present

    - name: Start Jenkins
      systemd:
        name: jenkins
        enabled: yes
        state: started

    - name: Start Docker
      systemd:
        name: docker
        enabled: yes
        state: started

    - name: Add Jenkins user to Docker group
      user:
        name: jenkins
        groups: docker
        append: yes

# =====================================================
# KUBERNETES MASTER + PROMETHEUS + GRAFANA
# =====================================================
- name: Setup Kubernetes Master
  hosts: k8s_master
  become: yes
  tasks:

    - name: Add Docker GPG key
      apt_key:
        url: https://download.docker.com/linux/ubuntu/gpg
        state: present

    - name: Add Docker repo
      apt_repository:
        repo: deb https://download.docker.com/linux/ubuntu {{ ansible_distribution_release }} stable
        state: present

    - name: Add Kubernetes GPG key
      apt_key:
        url: https://pkgs.k8s.io/core:/stable:/v1.29/deb/Release.key
        state: present

    - name: Add Kubernetes repo
      apt_repository:
        repo: deb https://pkgs.k8s.io/core:/stable:/v1.29/deb/ /
        state: present

    # Prometheus Repo
    - name: Add Prometheus repo
      apt_repository:
        repo: ppa:prometheus-community/prometheus
        state: present

    # Grafana Repo
    - name: Add Grafana GPG key
      apt_key:
        url: https://packages.grafana.com/gpg.key
        state: present

    - name: Add Grafana repo
      apt_repository:
        repo: deb https://packages.grafana.com/oss/deb stable main
        state: present

    - name: Update apt
      apt:
        update_cache: yes

    - name: Install Kubernetes + Monitoring
      apt:
        name:
          - docker-ce
          - docker-ce-cli
          - containerd.io
          - kubeadm
          - kubelet
          - kubectl
          - prometheus
          - grafana
        state: present

    - name: Start Docker
      systemd:
        name: docker
        enabled: yes
        state: started

    - name: Start Prometheus
      systemd:
        name: prometheus
        enabled: yes
        state: started

    - name: Start Grafana
      systemd:
        name: grafana-server
        enabled: yes
        state: started

    - name: Hold Kubernetes packages
      command: apt-mark hold kubeadm kubelet kubectl
